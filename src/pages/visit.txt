




import React, { useEffect, useState, useRef } from "react";
import { db } from "../firebase";
import {
  collection,
  addDoc,
  serverTimestamp,
  onSnapshot,
  query,
  orderBy,
  limit,
  getDocs,
} from "firebase/firestore";
import { Formik, Form, Field, ErrorMessage } from "formik";
import * as Yup from "yup";
import "./VisitForm.css";

function Visit() {
  // Helpers for default date/time in local timezone
  const pad = (n) => String(n).padStart(2, "0");
  const now = new Date();
  const defaultDate = `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())}`;
  const defaultTime = `${pad(now.getHours())}:${pad(now.getMinutes())}`;
  
  // Refs for animation targets
  const formRef = useRef(null);
  const successRef = useRef(null);

  // State management
  const [saving, setSaving] = useState(false);
  const [successMsg, setSuccessMsg] = useState("");
  const [visits, setVisits] = useState([]);
  const [agents, setAgents] = useState([]);
  const [showVisits, setShowVisits] = useState(false);

  // Initial form values
  const initialValues = {
    visitorName: "",
    phone: "",
    email: "",
    visitDate: defaultDate,
    visitTime: defaultTime,
    propertyTypes: [], // Changed to array for multiple selection
    referralSource: "",
    channelPartner: "", // Changed from brokerName to channelPartner
    channelPartnerPhone: "", // Added non-required field
    location: "",
    remarks: "",
    salesExecutive: "", // Changed from agentName to salesExecutive
    leadQuality: "", // Added lead quality field
  };

  // Validation schema
  const validationSchema = Yup.object({
    visitorName: Yup.string()
      .required("Visitor name is required"),
    phone: Yup.string()
      .matches(/^\d{10}$/, "Please enter a valid 10-digit phone number")
      .required("Contact number is required"),
    email: Yup.string()
      .email("Please enter a valid email address"),
    visitDate: Yup.date()
      .required("Visit date is required"),
    visitTime: Yup.string()
      .required("Visit time is required"),
    salesExecutive: Yup.string()
      .required("Sales Executive name is required"),
    propertyTypes: Yup.array()
      .min(1, "Please select at least one property type"),
    leadQuality: Yup.string()
      .required("Lead quality is required"),
  });

  // Realtime: last 10 visits list
  useEffect(() => {
    const q = query(
      collection(db, "siteVisits"),
      orderBy("createdAt", "desc"),
      limit(10)
    );
    const unsub = onSnapshot(q, (snap) => {
      const rows = snap.docs.map((d) => ({ id: d.id, ...d.data() }));
      setVisits(rows);
    });
    return unsub;
  }, []);

  // Fetch agent names for datalist suggestions
  useEffect(() => {
    getDocs(collection(db, "agents"))
      .then((snap) => {
        const names = snap.docs
          .map((d) => d.data()?.name)
          .filter(Boolean);
        setAgents(names);
      })
      .catch(() => {
        // If no agents collection, ignore silently
      });
  }, []);

  // Animation effect on load
  useEffect(() => {
    if (formRef.current) {
      formRef.current.classList.add('form-fade-in');
      setTimeout(() => {
        if (formRef.current) {
          formRef.current.classList.remove('form-fade-in');
        }
      }, 500);
    }
  }, []);

  // Handle property type selection
  const handlePropertyTypeToggle = (setFieldValue, values, type) => {
    const currentTypes = [...values.propertyTypes];
    const index = currentTypes.indexOf(type);
    
    if (index === -1) {
      // Add the type if not already selected
      currentTypes.push(type);
    } else {
      // Remove the type if already selected
      currentTypes.splice(index, 1);
    }
    
    setFieldValue("propertyTypes", currentTypes);
  };

  // Submit the form
  const handleSubmit = async (values, { resetForm }) => {
    // Show success message immediately when button is clicked
    setSuccessMsg("Visit form submitted");
    
    if (successRef.current) {
      successRef.current.classList.add('success-animation');
    }
    
    setSaving(true);
    try {
      const visitDateTime = new Date(`${values.visitDate}T${values.visitTime}:00`);
      const payload = {
        visitor: {
          name: values.visitorName.trim(),
          phone: values.phone,
          email: values.email.trim(),
          location: values.location.trim(),
        },
        propertyTypes: values.propertyTypes,
        channelPartner: {
          name: values.channelPartner.trim(),
          phone: values.channelPartnerPhone.trim(),
        },
        agent: {
          name: values.salesExecutive.trim(),
        },
        visitAt: visitDateTime,
        visitTime: values.visitTime,
        referralSource: values.referralSource,
        leadQuality: values.leadQuality,
        remarks: values.remarks.trim(),
        createdAt: serverTimestamp(),
      };

      await addDoc(collection(db, "siteVisits"), payload);
      
      // Reset form but keep salesExecutive name for convenience
      resetForm({
        values: {
          ...initialValues,
          salesExecutive: values.salesExecutive,
          visitTime: defaultTime, // Reset to current time
        }
      });
      
      setTimeout(() => {
        setSuccessMsg("");
        if (successRef.current) {
          successRef.current.classList.remove('success-animation');
        }
      }, 5000);
      
    } catch (error) {
      console.error("Error adding document: ", error);
      setSuccessMsg("Error submitting form. Please try again.");
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="visit-form-container">
      <div className="visit-form-header">
        <h1>Site Visit Form</h1>
        {/* <p>Schedule a visit to explore your dream property</p> */}
      </div>
      
      {successMsg && (
        <div className="success-message" ref={successRef}>
          <div className="success-icon">✓</div>
          <div className="success-text">
            <h3>Success!</h3>
            <p>{successMsg}</p>
          </div>
        </div>
      )}
      
      <div className="visit-form-card" ref={formRef}>
        <Formik
          initialValues={initialValues}
          validationSchema={validationSchema}
          onSubmit={handleSubmit}
        >
          {({ values, setFieldValue, errors, touched }) => (
            <Form>
              <div className="form-section">
                <h3 className="section-title">Visit Details</h3>
                
                <div className="form-row">
                  <div className="form-group">
                    <label htmlFor="visitDate">Visit Date*</label>
                    <Field
                      type="date"
                      id="visitDate"
                      name="visitDate"
                      className={errors.visitDate && touched.visitDate ? 'error' : ''}
                    />
                    <ErrorMessage name="visitDate" component="div" className="error-message" />
                  </div>
                  
                  <div className="form-group">
                    <label htmlFor="visitTime">Visit Time*</label>
                    <Field
                      type="time"
                      id="visitTime"
                      name="visitTime"
                      className={errors.visitTime && touched.visitTime ? 'error' : ''}
                      disabled={true} // Keep time fixed
                    />
                    <ErrorMessage name="visitTime" component="div" className="error-message" />
                  </div>
                </div>
                
                <div className="form-row">
                  <div className="form-group">
                    <label htmlFor="salesExecutive">Sales Executive*</label>
                    <Field
                      list="agent-names"
                      type="text"
                      id="salesExecutive"
                      name="salesExecutive"
                      placeholder="Select or type sales executive name"
                      className={errors.salesExecutive && touched.salesExecutive ? 'error' : ''}
                    />
                    <datalist id="agent-names">
                      {agents.map((a, i) => (
                        <option key={i} value={a} />
                      ))}
                    </datalist>
                    <ErrorMessage name="salesExecutive" component="div" className="error-message" />
                  </div>
                  
                  <div className="form-group">
                    <label htmlFor="leadQuality">Lead Status*</label>
                    <Field
                      as="select"
                      id="leadQuality"
                      name="leadQuality"
                      className={errors.leadQuality && touched.leadQuality ? 'error' : ''}
                    >
                      <option value="">Select Lead Status</option>
                      <option value="Hot">Hot</option>
                      <option value="Warm">Warm</option>
                      <option value="Cold">Cold</option>
                    </Field>
                    <ErrorMessage name="leadQuality" component="div" className="error-message" />
                  </div>
                </div>

                <div className="form-group full-width">
                  <label>Property Types*</label>
                  <div className="property-type-options">
                    {["1 BHK", "2 BHK", "3 BHK", "4 BHK", "PentHouse"].map((type) => (
                      <div key={type} className="property-type-option">
                        <label className={values.propertyTypes.includes(type) ? 'selected' : ''}>
                          <input
                            type="checkbox"
                            checked={values.propertyTypes.includes(type)}
                            onChange={() => handlePropertyTypeToggle(setFieldValue, values, type)}
                          />
                          {type}
                        </label>
                      </div>
                    ))}
                  </div>
                  {errors.propertyTypes && touched.propertyTypes && (
                    <div className="error-message">{errors.propertyTypes}</div>
                  )}
                </div>

                <div className="form-row">
                  <div className="form-group">
                    <label htmlFor="visitorName">Visitor Name*</label>
                    <Field
                      type="text"
                      id="visitorName"
                      name="visitorName"
                      placeholder="Enter visitor's full name"
                      className={errors.visitorName && touched.visitorName ? 'error' : ''}
                    />
                    <ErrorMessage name="visitorName" component="div" className="error-message" />
                  </div>
                  
                  <div className="form-group">
                    <label htmlFor="channelPartner">Channel Partner/Reference</label>
                    <Field
                      type="text"
                      id="channelPartner"
                      name="channelPartner"
                      placeholder="If applicable"
                    />
                  </div>
                </div>
                
                <div className="form-row">
                  <div className="form-group">
                    <label htmlFor="phone">Contact Number*</label>
                    <div className="phone-input-container">
                      <span className="phone-prefix">+91</span>
                      <Field
                        type="tel"
                        id="phone"
                        name="phone"
                        placeholder="10-digit number"
                        pattern="\d{10}"
                        inputMode="numeric"
                        className={errors.phone && touched.phone ? 'error' : ''}
                        onChange={(e) => {
                          // Only allow digits and limit to 10
                          const onlyDigits = e.target.value.replace(/\D/g, "").slice(0, 10);
                          setFieldValue("phone", onlyDigits);
                        }}
                      />
                    </div>
                    <ErrorMessage name="phone" component="div" className="error-message" />
                  </div>
                  
                  <div className="form-group">
                    <label htmlFor="channelPartnerPhone">Channel Partner Contact Number</label>
                    <div className="phone-input-container">
                      <span className="phone-prefix">+91</span>
                      <Field
                        type="tel"
                        id="channelPartnerPhone"
                        name="channelPartnerPhone"
                        placeholder="Optional"
                        pattern="\d{10}"
                        inputMode="numeric"
                        onChange={(e) => {
                          // Only allow digits and limit to 10
                          const onlyDigits = e.target.value.replace(/\D/g, "").slice(0, 10);
                          setFieldValue("channelPartnerPhone", onlyDigits);
                        }}
                      />
                    </div>
                  </div>
                </div>

                <div className="form-row">
                  <div className="form-group">
                    <label htmlFor="email">Email Address</label>
                    <Field
                      type="email"
                      id="email"
                      name="email"
                      placeholder="email@example.com"
                      className={errors.email && touched.email ? 'error' : ''}
                    />
                    <ErrorMessage name="email" component="div" className="error-message" />
                  </div>
                  
                  <div className="form-group">
                    <label htmlFor="location">Location/Address</label>
                    <Field
                      type="text"
                      id="location"
                      name="location"
                      placeholder="Visitor's location or property address"
                    />
                  </div>
                </div>

                <div className="form-group full-width">
  <label>Campaign Source</label>
  <div className="reference-options">
    {[
      "Online Search", 
      "Social Media", 
      "Friend/Family", 
      "Advertisement", 
      "Real Estate Portal", 
      "Hoardings/Billboards", 
      "Existing Client", 
      "Other"
    ].map((source) => (
      <div key={source} className="reference-option">
        <label className={values.referralSource.includes(source) ? 'selected' : ''}>
          <input
            type="checkbox"
            checked={values.referralSource.includes(source)}
            onChange={() => {
              const currentSources = Array.isArray(values.referralSource) 
                ? [...values.referralSource] 
                : [];
              const index = currentSources.indexOf(source);
              
              if (index === -1) {
                // Add the source if not already selected
                currentSources.push(source);
              } else {
                // Remove the source if already selected
                currentSources.splice(index, 1);
              }
              
              setFieldValue("referralSource", currentSources);
            }}
          />
          {source}
        </label>
      </div>
    ))}
  </div>
</div>
                
                <div className="form-group full-width">
                  <label htmlFor="remarks">Remarks</label>
                  <Field
                    as="textarea"
                    id="remarks"
                    name="remarks"
                    placeholder="Any additional notes or requirements"
                    rows="3"
                  />
                </div>
              </div>
              
              <div className="form-actions">
                <button 
                  type="submit" 
                  className="btn btn-submit" 
                  disabled={saving}
                >
                  {saving ? (
                    <>
                      <div className="spinner"></div>
                      Scheduling Visit...
                    </>
                  ) : (
                    <>
                      Schedule Visit
                      <svg viewBox="0 0 24 24" width="16" height="16" stroke="currentColor" strokeWidth="2" fill="none">
                        <path d="M5 13l4 4L19 7"/>
                      </svg>
                    </>
                  )}
                </button>
              </div>
            </Form>
          )}
        </Formik>
      </div>
    </div>
  );
}

export default Visit;
